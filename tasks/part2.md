Пример использования возможного С++ API:
```cpp
float* data_x{nullptr};
float* data_y{nullptr};
auto x_t = Tensor::from_blob(data_x, {.shape = {m, n}, .device = device_t::GPU, .dtype = dtype_t::Float32});
auto y_t = Tensor::from_blob(data_y, {.shape = {m, n}, .device = device_t::GPU, .dtype = dtype_t::Float32});
 
Variable x(x_t);     // graph node
Variable y(y_t);     // graph node
auto z = x * y;      // calculation graph
z.cpu();             // if there is no sum implementation for GPU
z.sum().backward();  // gradients calculation
x.grad();
y.grad();
```

Пример получения значений градиентов с использованием PyTorch:

```Python
import torch
 
x = torch.tensor([1,2,3], requires_grad=True)
y = torch.tensor([4,5,6], requires_grad=True)
z = x + y
z.sum().backward()
 
x.grad()
y.grad()
```
